;mode320
; Pixel test game: tests/pixel/check.py makes the images, runs this and checks the screenshots.

; a text window set in *define holds after game: the bottom 90 rows, halving what is under it
setwindow 8,160,18,3,16,16,0,2,0,0,1,#808080,0,150,319,239
game
*start
; lsp: opaque, alpha ramp, spfilter grey (kind 2) and grey of weights (kind 11)
lsp 1,":a;solid.png",10,10
lsp 2,":a;ramp.png",40,10
lsp 3,":a;solid.png",70,10
spfilter 3,2
lsp 4,":a;solid.png",100,10
spfilter 4,11,0,256,100,50,30
; a picture read in place from a Nitroplus NPA that cons_archives.txt names
lsp 5,":a;orig/bg/solid.png",130,10
; the same from a compressed NPA, one per kind of deflate block
lsp 6,":a;z/stored.png",175,10
lsp 7,":a;z/fixed.png",220,10
lsp 8,":a;z/dynamic.png",265,10
; a byte range of another file (a PNG inside a bank, as 最果てのイマ's)
lsp 9,":a;r/solid.png",10,60
; the same from a file XORed with a key by absolute offset (GAF4)
lsp 10,":a;r/xor.png",60,60
print 1
getscreenshot 320,240
savescreenshot "lsp.bmp"
csp -1
; lsp2: an 8x6 picture with a 1-pixel border at 100 %, 200 % and turned 90
lsp2 1,":a;border.png",40,40,100,100,0
lsp2 2,":a;border.png",100,40,200,200,0
lsp2 3,":a;border.png",160,40,100,100,90
print 1
getscreenshot 320,240
savescreenshot "lsp2.bmp"
csp2 -1
;spmotionclock num,-ms: a loop that jumps 50 px right at 1 s, started 50 s in, and the same from its start
lsp 11,":a;solid.png",10,40
spmotionclock 11,-50000
spmotion 11,1,0,0,0,100,100,0,256,0,1000,50,0,100,100,0,256,4,100000,50,0,100,100,0,256,0
lsp 12,":a;solid.png",10,80

print 1

savescreenshot "clock.bmp"
csp -1
print 1

bg "#c86432",10,1000
`x
getscreenshot 320,240
savescreenshot "window.bmp"
